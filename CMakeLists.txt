cmake_minimum_required(VERSION 3.21)
project(remarkable_learning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick Sql Test Network)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(remarkable_learning src/main.cpp src/InkCanvas.cpp src/PenInput.cpp src/EpaperRefresh.cpp src/ExerciseClient.cpp src/PracticeClient.cpp)
qt_add_qml_module(remarkable_learning URI RemarkableLearning VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(remarkable_learning PRIVATE Qt6::Quick Qt6::Sql Qt6::Network)
enable_testing()
qt_add_executable(ink_test tests/ink_test.cpp src/InkCanvas.cpp src/ExerciseClient.cpp src/PracticeClient.cpp)
target_link_libraries(ink_test PRIVATE Qt6::Quick Qt6::Sql Qt6::Test Qt6::Network)
add_test(NAME ink_test COMMAND ink_test)
qt_add_executable(pen_test tests/pen_test.cpp src/InkCanvas.cpp src/PenInput.cpp src/ExerciseClient.cpp src/PracticeClient.cpp)
target_link_libraries(pen_test PRIVATE Qt6::Quick Qt6::Sql Qt6::Test Qt6::Network)
add_test(NAME pen_test COMMAND pen_test)
set_tests_properties(ink_test pen_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=")
qt_add_executable(epaper_test tests/epaper_test.cpp src/EpaperRefresh.cpp)
target_link_libraries(epaper_test PRIVATE Qt6::Core Qt6::Test)
add_test(NAME epaper_test COMMAND epaper_test)
qt_add_executable(server_client_test tests/server_client_test.cpp src/ExerciseClient.cpp)
target_link_libraries(server_client_test PRIVATE Qt6::Core Qt6::Sql Qt6::Network Qt6::Test)
add_test(NAME server_client_test COMMAND server_client_test)

qt_add_executable(practice_client_test tests/practice_client_test.cpp src/PracticeClient.cpp)
target_link_libraries(practice_client_test PRIVATE Qt6::Core Qt6::Sql Qt6::Network Qt6::Test)
add_test(NAME practice_client_test COMMAND practice_client_test)
